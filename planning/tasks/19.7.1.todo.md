3. Dispatch sketch:
```rust
#[non_exhaustive]
#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub enum SoluteBackend { FiniteVolumeBackwardEuler }
impl SoluteBackend { pub const ALL: &'static [SoluteBackend] = &[SoluteBackend::FiniteVolumeBackwardEuler]; }

pub fn advance_solute_step(state, workspace, mesh, config, input) -> Result<(), SoluteStepError> {
    match config.backend() {
        SoluteBackend::FiniteVolumeBackwardEuler =>
            advance_finite_volume_backward_euler_step(state, workspace, mesh, config, input, None),
    }
}
pub fn advance_solute_step_with_processes(.., process_input) -> Result<..> {
    match config.backend() {
        SoluteBackend::FiniteVolumeBackwardEuler =>
            advance_finite_volume_backward_euler_step(.., process_input),
    }
}
fn advance_finite_volume_backward_euler_step(<old with_processes signature and body, verbatim>)
```
`#[non_exhaustive]` only affects other crates; inside the crate the match is exhaustive (no wildcard), so adding a variant later is a compile error here, which is the intent.
